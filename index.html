# pipeline real que roda no DETRAN-SP
def extrair(pdf: Path) -> dict:
    texto = ocr_em_cascata(pdf)      # PyMuPDF → Tesseract → RapidOCR
    texto = desgrudar(texto, "SEI", "PA")
    campos = {
        "nome":     nome_valido(texto),
        "registro": regex(r"\d{9,11}", texto),
        "sei":      regex(r"\d{3}\.\d{8}/\d{4}-\d{2}", texto),
        "pa":       regex(r"\d+/\d{4}", texto),
    }
    return {k: v for k, v in campos.items() if v}
